**Goal:** Build a Windows tray app that captures stereo meeting audio (mic L / WASAPI loopback R) and uploads it to `bae` server for transcription.

**Architecture:** Python tray app using PyAudioWPatch for dual-stream audio capture (WASAPI loopback + mic), pystray for system tray UI, and paramiko for atomic SFTP upload. Local staging directory ensures no data loss when the server is unreachable.

**Tech Stack:** Python 3.12, PyAudioWPatch, numpy, soundfile, pystray, Pillow, paramiko, pynput, tkinter

```python
import re
from bea_transcribe.config import SLUG_MAX_LENGTH


def slugify(name: str) -> str:
    """Convert meeting name to filesystem-safe slug."""
    if not name.strip():
        return "meeting"
    slug = name.lower()
    slug = re.sub(r"[^a-z0-9\s-]", "", slug)
    slug = re.sub(r"\s+", "-", slug.strip())
    slug = re.sub(r"-+", "-", slug)  # collapse consecutive hyphens
    slug = slug[:SLUG_MAX_LENGTH].rstrip("-")
    return slug or "meeting"
```

```python
from datetime import datetime


def make_job_name(start_time: datetime, meeting_name: str) -> str:
    """Generate job file basename: YYYY-MM-DD_HHmm_slug."""
    date_prefix = start_time.strftime("%Y-%m-%d_%H%M")
    return f"{date_prefix}_{slugify(meeting_name)}"


def create_metadata(
    meeting_name: str,
    recorded_by: str,
    start_time: datetime,
    end_time: datetime,
    sample_rate: int,
) -> dict:
    """Create metadata dict for the JSON sidecar file."""
    duration = int((end_time - start_time).total_seconds())
    return {
        "schema_version": 1,
        "meeting_name": meeting_name,
        "recorded_by": recorded_by,
        "start_time": start_time.isoformat(),
        "end_time": end_time.isoformat(),
        "duration_seconds": duration,
        "sample_rate": sample_rate,
        "channels": {"left": "bea", "right": "remote"},
    }
```

```python
"""Audio capture module: WASAPI loopback + mic → stereo FLAC."""

import numpy as np


def interleave_mono_to_stereo(
    left: np.ndarray, right: np.ndarray
) -> np.ndarray:
    """Interleave two mono arrays into a (N, 2) stereo array.

    Pads the shorter channel with zeros to match the longer one.
    """
    max_len = max(len(left), len(right))
    if max_len == 0:
        return np.empty((0, 2), dtype=np.float32)
    left_padded = np.zeros(max_len, dtype=np.float32)
    right_padded = np.zeros(max_len, dtype=np.float32)
    left_padded[: len(left)] = left
    right_padded[: len(right)] = right
    return np.column_stack((left_padded, right_padded))
```

```python
import logging
import queue
import threading
from datetime import datetime, timezone
from pathlib import Path

import pyaudiowpatch as pyaudio
import soundfile as sf

from bea_transcribe.config import STAGING_DIR

logger = logging.getLogger(__name__)

# Size of audio chunks in frames
CHUNK_FRAMES = 1024


def get_default_devices(pa: pyaudio.PyAudio) -> tuple[dict, dict]:
    """Return (mic_device_info, loopback_device_info).

    Finds the default WASAPI mic input and the loopback device
    for the default WASAPI output.
    """
    wasapi_info = pa.get_host_api_info_by_type(pyaudio.paWASAPI)
    default_output = pa.get_device_info_by_index(
        wasapi_info["defaultOutputDevice"]
    )
    default_input = pa.get_device_info_by_index(
        wasapi_info["defaultInputDevice"]
    )

    # Find the loopback device that matches the default output
    loopback = None
    for i in range(pa.get_device_count()):
        dev = pa.get_device_info_by_index(i)
        if dev.get("isLoopbackDevice") and dev["name"].startswith(
            default_output["name"].split(" (")[0]
        ):
            loopback = dev
            break

    if loopback is None:
        raise RuntimeError(
            f"No loopback device found for output: {default_output['name']}"
        )

    logger.info("Mic device: %s (rate=%d)", default_input["name"],
                int(default_input["defaultSampleRate"]))
    logger.info("Loopback device: %s (rate=%d)", loopback["name"],
                int(loopback["defaultSampleRate"]))

    return default_input, loopback


class AudioRecorder:
    """Records mic (left channel) and WASAPI loopback (right channel)
    into a stereo FLAC file."""

    def __init__(self):
        self._pa: pyaudio.PyAudio | None = None
        self._mic_stream = None
        self._loopback_stream = None
        self._mic_queue: queue.Queue[np.ndarray | None] = queue.Queue()
        self._loopback_queue: queue.Queue[np.ndarray | None] = queue.Queue()
        self._writer_thread: threading.Thread | None = None
        self._recording = False
        self._start_time: datetime | None = None
        self._output_path: Path | None = None
        self._sample_rate: int = 0

    @property
    def is_recording(self) -> bool:
        return self._recording

    @property
    def start_time(self) -> datetime | None:
        return self._start_time

    @property
    def sample_rate(self) -> int:
        return self._sample_rate

    def start(self) -> None:
        """Begin recording mic + loopback to a stereo FLAC in staging."""
        if self._recording:
            return

        self._pa = pyaudio.PyAudio()
        mic_info, loopback_info = get_default_devices(self._pa)

        # Use loopback's native sample rate for both streams
        self._sample_rate = int(loopback_info["defaultSampleRate"])
        mic_rate = int(mic_info["defaultSampleRate"])

        self._recording = True
        self._start_time = datetime.now(timezone.utc).astimezone()

        # Prepare output path in staging
        Path(STAGING_DIR).mkdir(parents=True, exist_ok=True)
        timestamp = self._start_time.strftime("%Y%m%d_%H%M%S")
        self._output_path = Path(STAGING_DIR) / f"recording_{timestamp}.flac"

        # Track if mic needs resampling
        self._mic_needs_resample = mic_rate != self._sample_rate
        self._mic_native_rate = mic_rate

        # Start writer thread first
        self._writer_thread = threading.Thread(
            target=self._writer_loop, daemon=True
        )
        self._writer_thread.start()

        # Loopback may be multi-channel (e.g., stereo output device).
        # Open at native channel count, downmix to mono in callback.
        self._loopback_channels = int(loopback_info["maxInputChannels"])
        logger.info("Loopback channels: %d", self._loopback_channels)

        # Open loopback stream at its native channel count
        self._loopback_stream = self._pa.open(
            format=pyaudio.paFloat32,
            channels=self._loopback_channels,
            rate=self._sample_rate,
            input=True,
            input_device_index=loopback_info["index"],
            frames_per_buffer=CHUNK_FRAMES,
            stream_callback=self._loopback_callback,
        )

        # Open mic stream at its native rate
        self._mic_stream = self._pa.open(
            format=pyaudio.paFloat32,
            channels=1,
            rate=self._mic_native_rate,
            input=True,
            input_device_index=mic_info["index"],
            frames_per_buffer=CHUNK_FRAMES,
            stream_callback=self._mic_callback,
        )

        logger.info("Recording started: %s (rate=%d)",
                     self._output_path, self._sample_rate)

    def stop(self) -> Path:
        """Stop recording and return the path to the FLAC file."""
        if not self._recording:
            raise RuntimeError("Not recording")

        self._recording = False

        # Stop streams
        if self._mic_stream:
            self._mic_stream.stop_stream()
            self._mic_stream.close()
        if self._loopback_stream:
            self._loopback_stream.stop_stream()
            self._loopback_stream.close()

        # Signal writer to finish
        self._mic_queue.put(None)
        self._loopback_queue.put(None)
        self._writer_thread.join(timeout=10)

        if self._pa:
            self._pa.terminate()
            self._pa = None

        logger.info("Recording stopped: %s", self._output_path)
        return self._output_path

    def _mic_callback(self, in_data, frame_count, time_info, status):
        if status:
            logger.warning("Mic callback status: %s", status)
        frames = np.frombuffer(in_data, dtype=np.float32)
        if self._mic_needs_resample:
            # Simple linear interpolation resample to match loopback rate
            ratio = self._sample_rate / self._mic_native_rate
            new_len = int(len(frames) * ratio)
            indices = np.linspace(0, len(frames) - 1, new_len)
            frames = np.interp(indices, np.arange(len(frames)), frames).astype(
                np.float32
            )
        self._mic_queue.put(frames)
        return (None, pyaudio.paContinue)

    def _loopback_callback(self, in_data, frame_count, time_info, status):
        if status:
            logger.warning("Loopback callback status: %s", status)
        frames = np.frombuffer(in_data, dtype=np.float32)
        # Downmix to mono if multi-channel (e.g., stereo loopback)
        if self._loopback_channels > 1:
            frames = frames.reshape(-1, self._loopback_channels).mean(axis=1)
        self._loopback_queue.put(frames.astype(np.float32))
        return (None, pyaudio.paContinue)

    def _writer_loop(self) -> None:
        """Drain both queues into buffers, write sample-aligned stereo to FLAC.

        Accumulates frames from each channel into separate buffers, then
        writes the minimum aligned length from both. This prevents L/R
        channel drift when callbacks fire at different cadences.
        """
        import time as _time

        mic_buf = np.array([], dtype=np.float32)
        loop_buf = np.array([], dtype=np.float32)

        with sf.SoundFile(
            str(self._output_path),
            mode="w",
            samplerate=self._sample_rate,
            channels=2,
            format="FLAC",
        ) as f:
            mic_done = False
            loop_done = False

            while not (mic_done and loop_done):
                # Drain all available chunks from mic queue
                while not mic_done:
                    try:
                        data = self._mic_queue.get_nowait()
                        if data is None:
                            mic_done = True
                        else:
                            mic_buf = np.concatenate([mic_buf, data])
                    except queue.Empty:
                        break

                # Drain all available chunks from loopback queue
                while not loop_done:
                    try:
                        data = self._loopback_queue.get_nowait()
                        if data is None:
                            loop_done = True
                        else:
                            loop_buf = np.concatenate([loop_buf, data])
                    except queue.Empty:
                        break

                # Write the aligned portion from both buffers
                min_len = min(len(mic_buf), len(loop_buf))
                if min_len > 0:
                    stereo = interleave_mono_to_stereo(
                        mic_buf[:min_len], loop_buf[:min_len]
                    )
                    f.write(stereo)
                    mic_buf = mic_buf[min_len:]
                    loop_buf = loop_buf[min_len:]

                if not (mic_done and loop_done):
                    _time.sleep(0.05)  # avoid busy-loop

            # Flush any remaining samples from whichever channel has more
            if len(mic_buf) > 0 or len(loop_buf) > 0:
                stereo = interleave_mono_to_stereo(mic_buf, loop_buf)
                f.write(stereo)
```

```python
"""SFTP upload with atomic .part rename and local staging."""

import logging
from pathlib import Path

import paramiko

from bea_transcribe.config import (
    QUEUE_PATH,
    SERVER_HOST,
    SERVER_USER,
    SSH_KEY_PATH,
    STAGING_DIR,
)

logger = logging.getLogger(__name__)


class StagingManager:
    """Manages the local staging directory for pending uploads."""

    def __init__(self, staging_dir: str | Path | None = None):
        self._dir = Path(staging_dir or STAGING_DIR)
        self._dir.mkdir(parents=True, exist_ok=True)

    @property
    def path(self) -> Path:
        return self._dir

    def pending_count(self) -> int:
        """Count pending jobs (each job = one .flac + one .json)."""
        return len(list(self._dir.glob("*.flac")))

    def pending_jobs(self) -> list[tuple[Path, Path]]:
        """Return list of (flac_path, json_path) pairs."""
        jobs = []
        for flac in sorted(self._dir.glob("*.flac")):
            json_f = flac.with_suffix(".json")
            if json_f.exists():
                jobs.append((flac, json_f))
        return jobs

    def remove_job(self, stem: str) -> None:
        """Remove a job's files from staging by stem name."""
        for suffix in (".flac", ".json"):
            path = self._dir / f"{stem}{suffix}"
            if path.exists():
                path.unlink()


def _connect_sftp() -> tuple[paramiko.SSHClient, paramiko.SFTPClient]:
    """Open SSH + SFTP connection to the server."""
    ssh = paramiko.SSHClient()
    ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())
    ssh.connect(
        SERVER_HOST,
        username=SERVER_USER,
        key_filename=SSH_KEY_PATH,
    )
    sftp = ssh.open_sftp()
    return ssh, sftp


def upload_job(flac_path: Path, json_path: Path) -> bool:
    """Upload a FLAC + JSON pair to bae with atomic .part rename.

    Returns True on success, False on failure.
    """
    flac_remote = f"{QUEUE_PATH}{flac_path.name}"
    json_remote = f"{QUEUE_PATH}{json_path.name}"
    flac_part = f"{flac_remote}.part"
    json_part = f"{json_remote}.part"

    ssh = None
    sftp = None
    try:
        ssh, sftp = _connect_sftp()

        # Upload both with .part suffix
        sftp.put(str(flac_path), flac_part)
        sftp.put(str(json_path), json_part)

        # Atomic rename both
        sftp.rename(flac_part, flac_remote)
        sftp.rename(json_part, json_remote)

        logger.info("Uploaded: %s", flac_path.stem)
        return True

    except Exception:
        logger.exception("Upload failed for %s", flac_path.stem)
        # Clean up any .part files left on server
        if sftp:
            for part in (flac_part, json_part):
                try:
                    sftp.remove(part)
                except Exception:
                    pass
        return False

    finally:
        if sftp:
            sftp.close()
        if ssh:
            ssh.close()


def upload_all_pending(staging: StagingManager) -> tuple[int, int]:
    """Try uploading all pending jobs. Returns (success_count, fail_count)."""
    success = 0
    fail = 0
    for flac, json_f in staging.pending_jobs():
        if upload_job(flac, json_f):
            staging.remove_job(flac.stem)
            success += 1
        else:
            fail += 1
    return success, fail
```

```python
"""System tray UI: icon, menu, meeting name dialog, global hotkey."""

import json
import logging
import threading
import tkinter as tk
from datetime import datetime, timezone
from pathlib import Path
from tkinter import messagebox

import pystray
from PIL import Image
from pynput import keyboard

from bea_transcribe.audio import AudioRecorder
from bea_transcribe.config import (
    HOTKEY,
    RECORDED_BY,
    STAGING_DIR,
)
from bea_transcribe.uploader import StagingManager, upload_all_pending
from bea_transcribe.util import create_metadata, make_job_name

logger = logging.getLogger(__name__)

ASSETS_DIR = Path(__file__).resolve().parent.parent.parent / "assets"


class TrayApp:
    """Main tray application coordinating recording, upload, and UI."""

    def __init__(self):
        self.recorder = AudioRecorder()
        self.staging = StagingManager()
        self._icon: pystray.Icon | None = None
        self._hotkey_listener: keyboard.GlobalHotKeys | None = None
        self._retry_timer: threading.Timer | None = None
        self._icon_idle = Image.open(str(ASSETS_DIR / "icon_idle.png"))
        self._icon_recording = Image.open(
            str(ASSETS_DIR / "icon_recording.png")
        )

    def run(self) -> None:
        """Start the tray icon and hotkey listener."""
        self._icon = pystray.Icon(
            "BEA-Transcribe",
            self._icon_idle,
            "BEA-Transcribe",
            menu=self._build_menu(),
        )
        self._start_hotkey_listener()
        self._schedule_retry()
        self._icon.run()

    def _build_menu(self) -> pystray.Menu:
        pending = self.staging.pending_count()
        if self.recorder.is_recording:
            return pystray.Menu(
                pystray.MenuItem("Stop Recording", self._on_stop),
                pystray.MenuItem("Exit", self._on_exit),
            )
        items = [pystray.MenuItem("Start Recording", self._on_start)]
        if pending > 0:
            items.append(
                pystray.MenuItem(
                    f"Upload Pending ({pending})", self._on_retry_upload
                )
            )
        items.append(pystray.MenuItem("Exit", self._on_exit))
        return pystray.Menu(*items)

    def _refresh_menu(self) -> None:
        if self._icon:
            self._icon.menu = self._build_menu()
            self._icon.update_menu()

    def _on_start(self, icon=None, item=None) -> None:
        try:
            self.recorder.start()
            if self._icon:
                self._icon.icon = self._icon_recording
                self._icon.title = "BEA-Transcribe (Recording...)"
            self._refresh_menu()
            logger.info("Recording started via UI")
        except Exception:
            logger.exception("Failed to start recording")
            if self._icon:
                self._icon.notify("Failed to start recording", "Error")

    def _on_stop(self, icon=None, item=None, auto_name: str | None = None) -> None:
        try:
            # Capture metadata before stop() in case stop clears it
            start_time = self.recorder.start_time
            sample_rate = self.recorder.sample_rate
            flac_path = self.recorder.stop()
            end_time = datetime.now(timezone.utc).astimezone()

            if self._icon:
                self._icon.icon = self._icon_idle
                self._icon.title = "BEA-Transcribe"
            self._refresh_menu()

            if auto_name:
                # Used by _on_exit to skip dialog
                self._finalize_recording(
                    flac_path, auto_name, start_time, end_time, sample_rate
                )
            else:
                # Show meeting name dialog in a new thread (tkinter needs its own)
                threading.Thread(
                    target=self._show_name_dialog,
                    args=(flac_path, start_time, end_time, sample_rate),
                    daemon=True,
                ).start()

        except Exception:
            logger.exception("Failed to stop recording")

    def _show_name_dialog(
        self, flac_path: Path, start_time: datetime,
        end_time: datetime, sample_rate: int,
    ) -> None:
        """Show a tkinter dialog to get the meeting name, then finalize."""
        root = tk.Tk()
        root.title("BEA-Transcribe — Save Recording")
        root.geometry("400x120")
        root.resizable(False, False)

        tk.Label(root, text="Meeting name:").pack(pady=(10, 0))
        entry = tk.Entry(root, width=50)
        today = datetime.now().strftime("%Y-%m-%d")
        entry.insert(0, today)
        entry.select_range(0, tk.END)
        entry.pack(pady=5)
        entry.focus_set()

        def on_ok(event=None):
            meeting_name = entry.get().strip() or today
            root.destroy()
            self._finalize_recording(
                flac_path, meeting_name, start_time, end_time, sample_rate
            )

        tk.Button(root, text="OK", command=on_ok, width=10).pack(pady=5)
        root.bind("<Return>", on_ok)
        root.mainloop()

    def _finalize_recording(
        self, flac_path: Path, meeting_name: str,
        start_time: datetime, end_time: datetime, sample_rate: int,
    ) -> None:
        """Rename recording, write metadata JSON, attempt upload."""
        job_name = make_job_name(start_time, meeting_name)
        staging_dir = Path(STAGING_DIR)
        staging_dir.mkdir(parents=True, exist_ok=True)

        # Rename FLAC to final name
        final_flac = staging_dir / f"{job_name}.flac"
        flac_path.rename(final_flac)

        # Write metadata JSON
        metadata = create_metadata(
            meeting_name, RECORDED_BY, start_time, end_time, sample_rate
        )
        json_path = staging_dir / f"{job_name}.json"
        json_path.write_text(json.dumps(metadata, indent=2))

        logger.info("Saved job: %s", job_name)

        # Attempt upload
        from bea_transcribe.uploader import upload_job

        if upload_job(final_flac, json_path):
            self.staging.remove_job(job_name)
            if self._icon:
                self._icon.notify(f"Uploaded: {meeting_name}", "Success")
        else:
            if self._icon:
                self._icon.notify(
                    f"Saved locally: {meeting_name}\nWill retry upload.",
                    "Upload Pending",
                )

        self._refresh_menu()

    def _on_retry_upload(self, icon=None, item=None) -> None:
        threading.Thread(target=self._do_retry, daemon=True).start()

    def _do_retry(self) -> None:
        success, fail = upload_all_pending(self.staging)
        if self._icon and success > 0:
            self._icon.notify(f"Uploaded {success} recording(s)", "Success")
        if self._icon and fail > 0:
            self._icon.notify(f"{fail} upload(s) still pending", "Retry Later")
        self._refresh_menu()

    def _schedule_retry(self) -> None:
        """Auto-retry pending uploads on a timer."""
        from bea_transcribe.config import RETRY_INTERVAL_SECONDS

        def retry_loop():
            if self.staging.pending_count() > 0:
                self._do_retry()
            self._schedule_retry()

        self._retry_timer = threading.Timer(
            RETRY_INTERVAL_SECONDS, retry_loop
        )
        self._retry_timer.daemon = True
        self._retry_timer.start()

    def _toggle_recording(self) -> None:
        if self.recorder.is_recording:
            self._on_stop()
        else:
            self._on_start()

    def _start_hotkey_listener(self) -> None:
        self._hotkey_listener = keyboard.GlobalHotKeys(
            {HOTKEY: self._toggle_recording}
        )
        self._hotkey_listener.start()

    def _on_exit(self, icon=None, item=None) -> None:
        if self.recorder.is_recording:
            # Auto-name with date to avoid blocking on dialog
            today = datetime.now().strftime("%Y-%m-%d")
            self._on_stop(auto_name=today)
        if self._hotkey_listener:
            self._hotkey_listener.stop()
        if self._retry_timer:
            self._retry_timer.cancel()
        if self._icon:
            self._icon.stop()
```

```python
"""BEA-Transcribe entry point."""

import logging
import sys
from logging.handlers import RotatingFileHandler
from pathlib import Path

from bea_transcribe.config import LOG_DIR


def setup_logging() -> None:
    log_dir = Path(LOG_DIR)
    log_dir.mkdir(parents=True, exist_ok=True)
    log_file = log_dir / "bea-transcribe.log"

    handler = RotatingFileHandler(
        log_file, maxBytes=5 * 1024 * 1024, backupCount=3
    )
    handler.setFormatter(
        logging.Formatter("%(asctime)s %(name)s %(levelname)s %(message)s")
    )

    root = logging.getLogger()
    root.setLevel(logging.INFO)
    root.addHandler(handler)


def main() -> None:
    setup_logging()
    logger = logging.getLogger(__name__)
    logger.info("BEA-Transcribe starting")

    try:
        from bea_transcribe.ui import TrayApp

        app = TrayApp()
        app.run()
    except Exception:
        logger.exception("Fatal error")
        # Show error dialog since we're running under pythonw (no console)
        try:
            import tkinter as tk
            from tkinter import messagebox

            root = tk.Tk()
            root.withdraw()
            messagebox.showerror(
                "BEA-Transcribe",
                "Failed to start. Check logs at:\n"
                f"{Path(LOG_DIR) / 'bea-transcribe.log'}",
            )
            root.destroy()
        except Exception:
            pass
        sys.exit(1)


if __name__ == "__main__":
    main()
```
